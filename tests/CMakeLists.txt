cmake_minimum_required(VERSION 3.16)
project(ilf_tests C)
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror)
add_subdirectory(../components/ilf_soul ilf_soul)
add_subdirectory(../components/ilf_herd ilf_herd)
add_executable(ilf_tests test_main.c)
target_link_libraries(ilf_tests ilf_soul m)
add_executable(ilf_herd_tests test_herd.c)
target_compile_definitions(ilf_herd_tests PRIVATE _GNU_SOURCE)
target_link_libraries(ilf_herd_tests ilf_herd m)
add_subdirectory(../components/ilf_phone ilf_phone)
add_executable(ilf_phone_tests test_phone.c)
target_link_libraries(ilf_phone_tests ilf_phone)
add_subdirectory(../components/ilf_ota ilf_ota)
add_executable(ilf_ota_tests test_ota.c)
target_link_libraries(ilf_ota_tests ilf_ota)
